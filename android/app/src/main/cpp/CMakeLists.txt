cmake_minimum_required(VERSION 3.22)
project(ArcadDwg C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LIBREDWG_LIBONLY ON CACHE BOOL "" FORCE)
set(LIBREDWG_DISABLE_WRITE ON CACHE BOOL "" FORCE)
set(LIBREDWG_DISABLE_JSON ON CACHE BOOL "" FORCE)
set(DISABLE_WERROR ON CACHE BOOL "" FORCE)
set(ENABLE_LTO OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(redwg_source URL "${CMAKE_CURRENT_LIST_DIR}/../../../../../vendor/libredwg-0.9.0-source.tar.gz" URL_HASH SHA256=54e5b1792340f50cc8cbf46e2796416a92f975e0060aea7821f2453f4c98bc74)
FetchContent_MakeAvailable(redwg_source)
add_library(arcad_dwg SHARED arcad_dwg.c)
target_include_directories(arcad_dwg PRIVATE ${redwg_source_SOURCE_DIR}/include ${redwg_source_SOURCE_DIR}/src ${redwg_source_BINARY_DIR}/src)
target_link_libraries(arcad_dwg redwg m log)
target_link_options(arcad_dwg PRIVATE "-Wl,-z,max-page-size=16384")
